## Appendix A — the measuring script

A scratch tool, not for commit. It builds with the project's own `vite.config.js` through
Vite's JS API and adds one plugin that dumps, for every output chunk, its static and
dynamic imports, its gzip size and each module's rendered length. Keep it outside the
repository and pass it the web root and an output path:

```bash
node build-measure.mjs /home/user/erg-dashboard/web /path/outside/the/repo/out.json
```

- It imports Vite from the web root's own `node_modules`, so it needs no install of its own.
- It writes `web/dist/`, which is ignored. Remove it afterwards if a later step should not
  mistake it for a fresh gate build.
- Rendered length is measured before minification. Use it to see which package is in which
  chunk, not to state a gzip figure.
- To see what production ships, set a placeholder `VITE_SENTRY_DSN` before running it (#465).

```js
// Orchestrator measurement: builds the app with the project's own vite.config.js and
// dumps per-chunk module attribution. Lives outside the repo; writes only to dist/ (ignored)
// and to the JSON path given.  usage: node build-measure.mjs <webRoot> <out.json>
import { writeFileSync } from 'node:fs';
import { gzipSync } from 'node:zlib';
import { pathToFileURL } from 'node:url';
import path from 'node:path';

const [webRoot, outJson] = process.argv.slice(2);
const { build } = await import(
  pathToFileURL(path.join(webRoot, 'node_modules/vite/dist/node/index.js')).href
);

const dump = {
  name: 'measure-dump',
  generateBundle(_opts, bundle) {
    const out = {};
    for (const [file, c] of Object.entries(bundle)) {
      if (c.type !== 'chunk') {
        const src = typeof c.source === 'string' ? Buffer.from(c.source) : Buffer.from(c.source);
        out[file] = { type: 'asset', bytes: src.length, gzip: gzipSync(src).length };
        continue;
      }
      const modules = {};
      for (const [id, m] of Object.entries(c.modules || {})) {
        modules[id] = m.renderedLength ?? (m.code ? m.code.length : null);
      }
      out[file] = {
        type: 'chunk',
        name: c.name,
        isEntry: c.isEntry,
        isDynamicEntry: c.isDynamicEntry,
        imports: c.imports,
        dynamicImports: c.dynamicImports,
        bytes: Buffer.byteLength(c.code),
        gzip: gzipSync(c.code).length,
        modules,
      };
    }
    writeFileSync(outJson, JSON.stringify(out));
  },
};

process.chdir(webRoot);
await build({
  root: webRoot,
  configFile: path.join(webRoot, 'vite.config.js'),
  logLevel: 'warn',
  plugins: [dump],
});
console.log('measure: wrote', outJson);
```
